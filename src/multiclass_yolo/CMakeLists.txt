cmake_minimum_required(VERSION 3.18)
project(multiclass_yolo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MULTICLASS_YOLO_TENSORRT "Build Jetson TensorRT + fused NV12 CUDA backend" ON)
find_package(ament_cmake REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/DepthaiWorkspace.cmake")
find_package(ament_index_cpp REQUIRED)
find_package(point_cloud REQUIRED)
find_package(roi_lidar REQUIRED)
find_package(oak_startup REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core dnn highgui imgproc)
# Initialize CUDA before creating its target: CUDA_ARCHITECTURES is copied
# from CMAKE_CUDA_ARCHITECTURES at target creation, not at target_sources().
if(MULTICLASS_YOLO_TENSORRT)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
endif()
add_library(multiclass_detector STATIC src/yolox_detector.cpp)
target_include_directories(multiclass_detector PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(multiclass_detector PUBLIC ${OpenCV_LIBS})
if(MULTICLASS_YOLO_TENSORRT)
  find_package(CUDAToolkit REQUIRED)
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h HINTS /usr/include/aarch64-linux-gnu /usr/include/x86_64-linux-gnu /usr/local/TensorRT/include)
  find_library(TENSORRT_LIBRARY nvinfer HINTS /usr/lib/aarch64-linux-gnu /usr/lib/x86_64-linux-gnu /usr/local/TensorRT/lib)
  find_library(TENSORRT_PARSER nvonnxparser HINTS /usr/lib/aarch64-linux-gnu /usr/lib/x86_64-linux-gnu /usr/local/TensorRT/lib)
  find_library(TENSORRT_PLUGIN nvinfer_plugin HINTS /usr/lib/aarch64-linux-gnu /usr/lib/x86_64-linux-gnu /usr/local/TensorRT/lib)
  if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY OR NOT TENSORRT_PARSER OR NOT TENSORRT_PLUGIN)
    message(FATAL_ERROR "TensorRT development headers/libraries required; see README.md")
  endif()
  target_sources(multiclass_detector PRIVATE src/tensorrt_yolox_backend.cpp src/cuda_yolox_preprocessor.cu)
  target_include_directories(multiclass_detector PRIVATE ${TENSORRT_INCLUDE_DIR})
  target_link_libraries(multiclass_detector PRIVATE CUDA::cudart ${TENSORRT_LIBRARY} ${TENSORRT_PARSER} ${TENSORRT_PLUGIN})
else()
  target_sources(multiclass_detector PRIVATE src/tensorrt_disabled.cpp)
endif()
add_executable(multiclass_yolo_node src/preview_node.cpp)
ament_target_dependencies(multiclass_yolo_node rclcpp ament_index_cpp)
target_link_libraries(multiclass_yolo_node multiclass_detector depthai::core Threads::Threads
  roi_lidar::roi_lidar_preview point_cloud::point_cloud_depth_source oak_startup::measurement)
install(TARGETS multiclass_yolo_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch models DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(FILES ../roi_vfh_test/config/reference/roi_lidar_test.yaml
  ../roi_vfh_test/config/reference/bev_config_test.yaml DESTINATION share/${PROJECT_NAME}/config/vfh_reference)
if(BUILD_TESTING)
  enable_testing()
  add_executable(multiclass_preview_exit_test test/preview_exit_test.cpp)
  target_include_directories(multiclass_preview_exit_test PRIVATE include)
  add_test(NAME multiclass_preview_exit COMMAND multiclass_preview_exit_test)
  add_executable(multiclass_postprocess_test test/postprocess_test.cpp)
  target_link_libraries(multiclass_postprocess_test multiclass_detector)
  add_test(NAME multiclass_postprocess COMMAND multiclass_postprocess_test)
  add_executable(multiclass_object_localization_test test/object_localization_test.cpp)
  target_include_directories(multiclass_object_localization_test PRIVATE include)
  add_test(NAME multiclass_object_localization COMMAND multiclass_object_localization_test)
  add_executable(multiclass_model_smoke test/model_smoke.cpp)
  target_link_libraries(multiclass_model_smoke multiclass_detector)
  add_test(NAME multiclass_model_smoke COMMAND multiclass_model_smoke "${CMAKE_CURRENT_SOURCE_DIR}/models/multiclass_yolox_tiny_512x320_holdout10.onnx")
endif()
ament_package()
