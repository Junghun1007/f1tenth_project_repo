cmake_minimum_required(VERSION 3.16)
project(roi_lidar)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(point_cloud REQUIRED)
find_package(oak_startup REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui)
find_package(Threads REQUIRED)
add_library(roi_lidar_preview STATIC src/preview.cpp)
target_include_directories(roi_lidar_preview PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(roi_lidar_preview PUBLIC point_cloud::point_cloud_depth_source ${OpenCV_LIBS})
add_executable(roi_lidar_node src/roi_lidar_node.cpp)
ament_target_dependencies(roi_lidar_node rclcpp rcl_interfaces sensor_msgs std_msgs)
target_link_libraries(roi_lidar_node roi_lidar_preview oak_startup::measurement Threads::Threads)
install(TARGETS roi_lidar_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(roi_lidar_config_test test/config_test.cpp)
  target_include_directories(roi_lidar_config_test PRIVATE include)
  target_link_libraries(roi_lidar_config_test point_cloud::point_cloud_depth_source)
  add_test(NAME roi_lidar_config_test COMMAND roi_lidar_config_test)
  add_executable(roi_lidar_scan_test test/scan_test.cpp)
  target_include_directories(roi_lidar_scan_test PRIVATE include)
  target_link_libraries(roi_lidar_scan_test point_cloud::point_cloud_depth_source)
  add_test(NAME roi_lidar_scan_test COMMAND roi_lidar_scan_test)
  add_executable(roi_lidar_preview_test test/preview_test.cpp)
  target_link_libraries(roi_lidar_preview_test roi_lidar_preview)
  add_test(NAME roi_lidar_preview_test COMMAND roi_lidar_preview_test)
endif()
ament_package()
