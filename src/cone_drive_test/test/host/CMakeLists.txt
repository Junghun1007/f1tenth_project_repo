# ROS 없는 호스트 빌드: core + 순수 CPU 복사본 + 테스트
#   cmake -S src/cone_drive_test/test/host -B build/host && cmake --build build/host -j4 && ctest --test-dir build/host --output-on-failure
cmake_minimum_required(VERSION 3.16)
project(cone_drive_test_host CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(PKG ${CMAKE_CURRENT_LIST_DIR}/../..)
include(${PKG}/cmake/core_sources.cmake)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc dnn)
find_package(Threads REQUIRED)
add_library(cdt_core STATIC ${CDT_CORE_SOURCES} ${CDT_WEB_SOURCES})
target_include_directories(cdt_core PUBLIC ${PKG}/include ${PKG}/third_party ${OpenCV_INCLUDE_DIRS})
target_link_libraries(cdt_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
target_compile_definitions(cdt_core PUBLIC ${CDT_WEB_DEFINITIONS})
target_compile_options(cdt_core PRIVATE -Wall -Wextra)
enable_testing()
file(GLOB CDT_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/*_test.cpp)
foreach(test_src ${CDT_TESTS})
  get_filename_component(name ${test_src} NAME_WE)
  add_executable(${name} ${test_src} ${CMAKE_CURRENT_LIST_DIR}/check_main.cpp)
  target_link_libraries(${name} cdt_core)
  target_compile_definitions(${name} PRIVATE CDT_FIXTURES="${PKG}/test/fixtures" CDT_MODELS="${PKG}/models" CDT_WEB="${PKG}/web")
  add_test(NAME ${name} COMMAND ${name})
endforeach()
# 원본 패키지 테스트 (자체 main): multiclass_yolo, camera_driver
add_library(cdt_tensorrt_disabled STATIC ${CDT_DETECTOR_SOURCES} ${PKG}/src/copied/tensorrt_disabled.cpp)
target_link_libraries(cdt_tensorrt_disabled cdt_core)
file(GLOB CDT_ORIGINAL_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/original/*.cpp)
foreach(test_src ${CDT_ORIGINAL_TESTS})
  get_filename_component(name ${test_src} NAME_WE)
  add_executable(original_${name} ${test_src})
  target_link_libraries(original_${name} cdt_core cdt_tensorrt_disabled)
endforeach()
add_test(NAME original_postprocess COMMAND original_postprocess_test)
add_test(NAME original_model_smoke COMMAND original_model_smoke ${PKG}/models/multiclass_yolox_tiny_512x320_holdout10.onnx)
add_test(NAME original_imu_image_stabilizer COMMAND original_imu_image_stabilizer_test)
add_test(NAME original_vehicle_motion_estimator COMMAND original_vehicle_motion_estimator_test)
