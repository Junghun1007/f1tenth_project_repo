cmake_minimum_required(VERSION 3.16)
project(roi_vfh_test)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(line_detactor REQUIRED)
find_package(auto_control REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui)
add_executable(roi_vfh_preview src/preview_node.cpp)
target_include_directories(roi_vfh_preview PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(roi_vfh_preview rclcpp rcl_interfaces nav_msgs sensor_msgs std_msgs geometry_msgs line_detactor)
target_link_libraries(roi_vfh_preview auto_control::control_core ${OpenCV_LIBS})
install(TARGETS roi_vfh_preview DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME roi_vfh_launch_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/launch_test.py)
  add_executable(roi_vfh_render_test test/render_test.cpp)
  target_include_directories(roi_vfh_render_test PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(roi_vfh_render_test ${OpenCV_LIBS})
  add_test(NAME roi_vfh_render_test COMMAND roi_vfh_render_test)
  foreach(name planner motion)
    add_executable(roi_vfh_${name}_test test/${name}_test.cpp)
    target_include_directories(roi_vfh_${name}_test PRIVATE include)
    add_test(NAME roi_vfh_${name}_test COMMAND roi_vfh_${name}_test)
  endforeach()
endif()
ament_package()
