cmake_minimum_required(VERSION 3.16)
project(stereo_rgb_sweep LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(depthai 3.6 CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(camera_probe camera_probe.cpp)
target_link_libraries(camera_probe PRIVATE depthai::core Threads::Threads)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/depthai_runtime_dir.txt"
  CONTENT "$<TARGET_FILE_DIR:depthai::core>\n")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(camera_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
